}
</style>


<body>
    <div id="ceshi">
        这里是显示复制内容
    </div>
    
    <div>
        <button onclick="copy()">复制</button>
    </div>
    <!-- 此处为一个隐藏dom  注：不能直接使用display:none -->
    <div style="opacity: 0;">
        <input id="test" type="text" value="这是一个测试">
        <!-- value 值绑定为你需要复制的值即可 -->
    </div>

    <script src="js/jquery.min.js"></script>
    <script>
        function copy() {
            var copyDom = $("#test");
            copyDom.select(); // 选择对象

            document.execCommand("Copy"); // 执行浏览器复制命令

            var copyTxt = $("#test").val();
           
            copyDom.style("display","none")//复制成功后删除input元素   如需多次使用则不隐藏

            $("#ceshi").text(copyTxt)//复制成功后插入

        }
    </script>
</body>

</html>
