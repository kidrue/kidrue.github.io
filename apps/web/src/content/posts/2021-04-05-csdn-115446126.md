export function filterThisRoutes(routes, roles) {
  const res = [];

  slideIndex = Cookie.get('sliIndex');
  // slideIndex = store.state.user.slideIndex;
  const slideList = [homeRoutes, extendRounts, rechargeRounts, committedRounts, accountdRounts]
  // toLowerCase() 
  for (let i = 0; i < slideList[slideIndex].length; i++) {
    if (slideList[slideIndex][i].meta == undefined) {
      res.push(slideList[slideIndex][i])
    } else {
      for (let j = 0; j < roles.length; j++) {
        var slideInfo;
        if (slideList[slideIndex][i].meta.roles[0].toLowerCase() == roles[j].toLowerCase()) {
          slideInfo = slideList[slideIndex][i];
          if (slideInfo.children != undefined) {
            var childrenInfo = [];
            for (let y = 0; y < slideInfo.children.length; y++) {
              if (slideInfo.children[y].meta == undefined) {
                childrenInfo.push(slideInfo.children[y])
              } else {
                for (let x = 0; x < roles.length; x++) {
                  if (slideInfo.children[y].meta.roles[0].toLowerCase() == roles[x].toLowerCase()) {
                    childrenInfo.push(slideInfo.children[y]);
                  }
                }
              }
            }
            slideInfo.children = childrenInfo;
          }
          res.push(slideInfo)
        }
      }

    }
  }
  return res
}
