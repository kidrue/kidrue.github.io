var dialogMgr = {
    isshow: false,
    openindex: 0,

    /*关闭弹出窗体*/
    closeDialog: function(dialogID) {
        api.closeFrame({
            name: 'frame_my_dialog'
        });
        dialogMgr.isshow = false;
        if(dialogID != -1)
            dialogList[dialogID].ischeck = false;
    },

    /* 打开弹出框 */
    //dialog_id: 窗体i的ID
    openDialog: function(url,dialog_id, parmes) {
        api.openFrame({
            name: 'frame_my_dialog',
            url: url,
            bgColor: 'rgba(0,0,0,0.5)',
            rect: {
                x: 0,
                y: 0,
                w: api.winWidth,
                h: api.winHeight
            },
            animation: {
                type: "movein",
                subType: "from_top",
                duration: 50
            },
            pageParam: {
                dialogid:dialog_id,
                datainfo: parmes,
            },
            bounces: false,
        });


        dialogMgr.isshow = true;
    },
};
