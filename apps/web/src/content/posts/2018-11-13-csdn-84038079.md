该代码用于使连接该机ip相应端口使做出答复

```
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.io.OutputStreamWriter;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.ArrayList;

import com.test.zq11_9.Ceshi;

public class Test2 extends Thread {
	public Socket client = new Socket();//创建socket类对象套接字接口client,为了可以传入run方法
	private static ArrayList<Socket> list=new ArrayList<Socket>();//将客户机
	
	public Test2(Socket client) {
		this.client = client;
	}

	public void run() {
		
		try {
			OutputStream out = client.getOutputStream();//将传入数据传入输出流out

			out.write("你好".getBytes());//在客户机上输出"你好"
			InputStream in = client.getInputStream();//读取客户机上的输入值
			InputStreamReader isr = new InputStreamReader(in);//创建字符输入流类对象isr,并将in对象传入isr

			BufferedReader br = new BufferedReader(isr);//isr值传入更高效率buffered类
			String msg = "";
			while ((msg = br.readLine()) != null) {
				for(int i=0;i<list.size();i++){
				Socket socket=list.get(i);//新建套接字端口socket,并将集合里的每个用户传入socket进行以下判断
				if(!socket.equals(client)){//将消息发给非client的客户机
					OutputStream os=socket.getOutputStream();
					os.write((msg+"\r\n").getBytes());//将数据发给每个socket对象并进行换行
				}
				}
			}//
		} catch (IOException e) {
			e.printStackTrace();
		}
	}

	
	}
}
```
