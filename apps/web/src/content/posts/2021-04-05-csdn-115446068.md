var parentInfo = [{

/**
 *  利用高德api获取行政区边界geoJson
 *   adcode 行政区code 编号
 **/

//此版本不再维护，准备在写另一个新版本

function getGeoJson(adcode) {
    AMapUI.loadUI(['geo/DistrictExplorer'], function(DistrictExplorer) {
        var districtExplorer = new DistrictExplorer()
        districtExplorer.loadAreaNode(adcode, function(error, areaNode) {
            if (error) {
                console.error(error);
                return false;
            }
            let Json = areaNode.getSubFeatures();
            if (Json.length > 0) {
                parentJson = Json
            } else if (Json.length === 0) {
                Json = parentJson.filter(function(item) {
                    if (item.properties.adcode == adcode) {
                        return item
                    }
                })
                if (Json.length === 0) return
            }
            // 获取当前所有地区编码
            for (var i = 0; i < Json.length; i++) {
                pageMol.dqCodeList[i] = Json[i].properties.adcode;
            }
            if (adcode == 100000) {
              adcode = ""
            }
            shopInfo(Json,adcode);
        });
    })
}

var shopvalue = 0;
var uservalue = 0;
//获取数据，这里我们用随机数模拟数据

function getMapData(Json) {
    var mapData = Json.map(function(item) {
        shopvalue = 0;
        uservalue = 0;
        for (var j = 0; j < cityInfo.length; j++) {
            if (item.properties.adcode == cityInfo[j].cityCode) {
                shopvalue = cityInfo[j].shop_num;
                uservalue = cityInfo[j].area_num;
            }
        }
        return ({
            name: item.properties.name,
            value: uservalue * 10,
            shopvalue: shopvalue,
            uservalue: uservalue,
            level: item.properties.level,
            cityCode: item.properties.adcode
        })
    });

    var pointData = [];
    var pointData = Json.map(function(item) {
        return ({
            name: item.properties.name,
            value: [item.properties.center[0], item.properties.center[1], 50],
           cityCode: item.properties.adcode
        })
    })
    let mapJson = {}
        //geoJson必须这种格式
    mapJson.features = Json

    //去渲染echarts
    initEcharts(mapData, pointData, mapJson)
}

function initEcharts(mapData, pointData, mapJson) {

    echarts.registerMap('Map', mapJson);

    //这里加true是为了让地图重新绘制，不然如果你有筛选的时候地图会飞出去
    myChart.setOption({

        globalCoord: false, // 缺省为 false

            trigger: "item",

                let shopvalue = p.data.shopvalue;
                let uservalue = p.data.uservalue;

                    shopvalue = 0;

                if (window.isNaN(uservalue)) {
                    uservalue = 0;
                }
                let txtCon =
                    p.name + "<br>" + "<hr>" + "用户人数 : " + uservalue + "<br>" + "<hr>" + "商户数量 : " + shopvalue;
                return txtCon;
            }
        },
        visualMap: {
            type: 'piecewise',
            pieces: [{
                max: 0,
                label: '安全',
                color: 'green'
            }, {
                min: 0,
                max: 50,
                label: '警告',
                color: '#d08a00'
            }, {
                min: 100,
                label: '危险',
                color: '#c23c33'

            textStyle: {
                color: '#24CFF4'
            },
            visibility: 'off'
        },
        series: [{
            name: "地图",
            type: "map",
            map: "Map",
            roam: true, //是否可缩放

            data: mapData,

                normal: {
                    show: true,

                    borderWidth: '1',

                                p.name = "新疆"

                            case '香港特别行政区':
                                p.name = "香港"
                                break;
                            case '澳门特别行政区':
                                p.name = "澳门"

                        let txtCon =

//echarts点击事件

function echartsMapClick(params) {
    //如果当前是最后一级，那就直接return
    if (parentInfo[parentInfo.length - 1].code == params.data.cityCode) {
        return
    }
    level = level + 1;
    let data = params.data
    parentInfo.push({
        cityName: data.name,
        level: data.level,
        code: data.cityCode
    })
    getGeoJson(data.cityCode)
}

// 区域信息
function shopInfo(Json,code) {
    pageMol.dataitem = [];
    var parms = {
        url: "Agency/newlist",
        values: {
            ty: level,
            Code:code
        },
        callback: function(ret) {
            if (ret.flag) {
                cityInfo = ret.data.list;
                pageMol.dataitem[0] = {
                    avatarUrl: "代理头像",
                    nickName: "代理昵称",
                    phone1: "手机号",
                    name: "所在地区",
                    area_num: "用户数量",
                    shop_num: "商户数量",
                    adcode: 0,
                }
                for (var i = 0; i < ret.data.list.length; i++) {
                    var info = ret.data.list[i];
                    pageMol.dqCodeList.map(function(item) {
                        if (item == info.cityCode && info.phone != "") {
                            info = clInfo(info)
                            pageMol.dataitem.push(info);
                        }

                        return true;
                    })
                }
                //去获取数据
                getMapData(Json)
            }
        },
    };
    JM_POST(parms);
}

function clInfo(info) {
    if (info.avatarUrl == "") {
        info.avatarUrl = "../image/jm_logo.png"
    }
    var reg = /^(\d{3})\d+(\d{3})$/;
    if (info.phone != null && info.phone != 0) {
        info.phone1 = info.phone.replace(reg, "$1****$2");
    }
    return info
}
