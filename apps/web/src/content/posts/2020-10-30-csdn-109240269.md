```javascript

                    //直接通过displayed的布尔值直接使点击无效
                    displayed=false;
                    const countDown = setInterval(() => {
                    if (count == 0) {
                        displayed=true;
                        $('#fasong').text('重新发送');
                        $(this).removeClass(".active");
                        clearInterval(countDown);
                    } else {
                        $('#fasong').text(count + '秒后可重新获取');
                        $(this).addClass(".active");
                    }
                    count--;
                }, 1000);
```
