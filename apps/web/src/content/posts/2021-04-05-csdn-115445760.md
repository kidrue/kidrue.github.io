```javascript
      // 下载app
        function downAPP() {
          var systemType = api.systemType;
            if (systemType == "android") {
                $api.text($api.dom("#update"), "");
                api.download({
                    url: pageMol.updateInfo.android.down_path,
                    savePath: 'fs://' + GenNonDuplicateID() + '.apk',
                    report: true,
                    cache: true,
                    allowResume: true
                }, function(ret, err) {
                    console.log(ret.percent);
                    $("#probar").animate({
                        width: ret.percent + "%",
                    }, 300);
                    $api.text($api.dom(".gx"), ret.percent + "%");
                    if (ret.state == 1) {
                        $api.text($api.dom(".gx"), "");
                        $("#install").toggle();
                        // $api.css($api.dom(".lx"), 'background-image:linear-gradient(to right,#eee,#eee);color:#333;');
                        pageMol.appDownload = ret.savePath
                        toast('下载完成\n' + "大小：" + ret.fileSize + '\n' + '保存路径：' + ret.savePath);
                        installAPP();
                    }
                });
            } else {
                installAPP();
            }
        }
            // 安装app
        function installAPP() {
            var systemType = api.systemType;
            console.log(pageMol.appDownload)
            if (systemType == "android") {
                api.installApp({
                    appUri: pageMol.appDownload
                });
            } else {
                api.installApp({
                    appUri: pageMol.updateInfo.ios.down_path
                });
              /*  api.closeFrame({
                    name: 'frame_updateApp'
                });*/
            }
        }
```
