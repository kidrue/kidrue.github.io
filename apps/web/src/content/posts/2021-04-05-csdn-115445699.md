  function canvasGoodsImg(ewm, info, callback) {
            //显示合成图片loading
            api.showProgress({
                title: '',
                text: '合成图片',
                modal: false
            });
            var Tit = info.goods_name;
            if (Tit.length > 31) {
                Tit = Tit.substring(0, 31);
            }
            var canvas = document.getElementById("sharecanvas");
            var ctx = canvas.getContext("2d");
            canvas.height = 700;
            canvas.width = 600;
            // 商品图
            var shopImg = new Image();
            shopImg.setAttribute('crossOrigin', 'anonymous');
            shopImg.src = info.shareImg;
            shopImg.onload = function() {
                // 设置背景色
                ctx.fillStyle = '#ffffff';
                ctx.fillRect(0, 0, canvas.width, canvas.height);
                // 设置商品图
                shopImg.width = 600;
                shopImg.height = 600;
                ctx.drawImage(shopImg, 0, 0, 750, 625, 0, 0, 600, 500);
                // 二维码
                var img = new Image();
                img.src = ewm;
                img.setAttribute('crossOrigin', 'anonymous');
                img.onload = function() {
                    img.width = 600;
                    img.height = 600;
                    ctx.drawImage(img, 0, 500, 200, 200);
                    // 商品标题
                    var chr = Tit.split("");
                    var temp = "";
                    var row = [];
                    for (var a = 0; a < chr.length; a++) {
                        if (ctx.measureText(temp).width < 130) {;
                        } else {
                            row.push(temp);
                            temp = "";
                        }
                        temp += chr[a];
                    }
                    row.push(temp);
                    for (var b = 0; b < row.length; b++) {
                        ctx.font = "26px Arial";
                        ctx.lineWidth = 1;
                        ctx.fillStyle = '#999999';
                        ctx.fillText(row[b], 210, 500 + (b + 1) * 40);
                    }
                    //  商品现价
                    ctx.font = "36px Arial";
                    ctx.lineWidth = 1;
                    ctx.fillStyle = '#999999';
                    ctx.fillText("￥" + info.goods_price, 420, 660);
                    // 商品原价
                    ctx.font = "26px Arial";
                    ctx.lineWidth = 1;
                    ctx.fillStyle = '#999999';
                    ctx.fillText("￥" + info.line_price, 300, 660);
                    // 删除线
                    ctx.strokeStyle = '#aba0ac';
                    ctx.lineWidth = 1; // 改变线的粗细
                    ctx.moveTo(300, 650); // 起始点
                    ctx.lineTo(420, 650); // 第二点(如果是一条直线的话，就是终点)
                    ctx.stroke();
                    // 保存图片
                    var imagedata = canvas.toDataURL("img/png");
                    // base64转换并保存在设备本地    图片格式
                    var trans = api.require('trans');
                    imagedata = imagedata.replace('data:image/png;base64,', '');
                    trans.saveImage({
                        base64Str: imagedata,
                        imgPath: "fs://jmimg/",
                        imgName: info.goods_id + "_jm.png",
                    }, function(ret, err) {
                        if (ret.status) {
                            $("#sharecanvas").css("display", "block")
                            console.log("保存合成图:" + JSON.stringify(ret));
                            imgurl = "fs://jmimg/" + info.goods_id + "_jm.png";
                        } else {
                            $("#sharecanvas").css("display", "block")
                            imgurl = "fs://jmimg/" + info.goods_id + "_jm.png";
                        }
                    });
                }
            }
             //隐藏合成图片loading
            api.hideProgress();
        }
